 COMMUNITY CRM — RUBRIC AND DATA NOTES
 Data storage: `db` is the live in-memory application state. `loadDB()` reads a JSON
 snapshot from localStorage on startup; `saveDB()` writes changes back to localStorage.
 This makes demo records survive a browser refresh on the same browser/device. It is
 not a remote database, and clearing browser storage removes the saved data.
 All CRUD functions below update `db` first and then call saveDB() and render().
 For production, replace demo authentication and localStorage with a secure backend.
*/

function addYears(date,years){const d=new Date((date||today())+"T12:00:00");d.setFullYear(d.getFullYear()+Number(years||1));return d.toISOString().slice(0,10)}
function loadDB(){try{const saved=localStorage.getItem(STORAGE_KEY);if(saved){const parsed=JSON.parse(saved);if(parsed.contacts&&parsed.households&&parsed.memberships&&parsed.donations&&parsed.tasks)return parsed;}}catch(e){console.warn("Could not read saved CRM data",e)}return seedDB()}

function saveDB(){try{localStorage.setItem(STORAGE_KEY,JSON.stringify(db));return true}catch(e){alert("Could not save to this browser. Storage may be full or disabled.");return false}}

function parseCSV(text){
 // Small RFC4180-style parser: handles commas and escaped quotes inside quoted fields.
 const rows=[];let row=[],field="",quoted=false;
 for(let i=0;i<text.length;i++){const ch=text[i];if(quoted){if(ch==='"'&&text[i+1]==='"'){field+='"';i++}else if(ch==='"')quoted=false;else field+=ch}
 else if(ch==='"')quoted=true;else if(ch===","){row.push(field);field=""}else if(ch==="\n"){row.push(field.replace(/\r$/,""));rows.push(row);row=[];field=""}else field+=ch}
 if(field.length||row.length){row.push(field.replace(/\r$/,""));rows.push(row)}
 if(quoted)throw new Error("CSV has an unclosed quotation mark.");
 return rows.filter(r=>r.some(v=>v.trim()!==""));
}
async function importCSV(file){
 const output=document.getElementById("importResult");if(!file)return;
 try{
  const rows=parseCSV(await file.text());if(rows.length<2)throw new Error("CSV needs a header row and at least one data row.");
  const headers=rows[0].map(h=>h.trim());const required=["firstname","lastname"];
  if(!required.every(h=>headers.some(x=>x.toLowerCase()===h)))throw new Error("Required CSV columns: firstName and lastName.");
  const idx={};headers.forEach((h,i)=>idx[h.toLowerCase()]=i);let added=0,skipped=0,invalid=0;
  rows.slice(1).forEach((r,n)=>{
   const get=k=>(r[idx[k]]||"").trim(),firstName=get("firstname"),lastName=get("lastname"),email=get("email");
   if(!firstName||!lastName){invalid++;return}
   if(email&&!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)){invalid++;return}
   if(email&&db.contacts.some(c=>c.email.toLowerCase()===email.toLowerCase())){skipped++;return}
   const houseName=get("household");let household="";
   if(houseName){let h=db.households.find(x=>x.name.toLowerCase()===houseName.toLowerCase());if(!h){h={id:uid("h"),name:houseName,address:"",notes:"Imported from CSV"};db.households.push(h)}household=h.id}
   const c={id:uid("c"),firstName,lastName,email,phone:get("phone"),type:get("type")||"Prospect",status:get("status")==="Archived"?"Archived":"Active",household,notes:"Imported from CSV",created:today(),history:[{date:today(),text:"Imported from CSV"}]};
   db.contacts.push(c);added++;
  });
  if(added){logActivity(`CSV import added ${added} contact(s); skipped ${skipped} duplicate(s); rejected ${invalid} invalid row(s).`);saveDB()}
  output.className="notice "+(invalid?"error":"success");output.textContent=`Import finished: ${added} added, ${skipped} duplicate(s) skipped, ${invalid} invalid row(s) rejected.`;render();
  const out2=document.getElementById("importResult");if(out2){out2.className=output.className;out2.textContent=output.textContent}
 }catch(e){output.className="notice error";output.textContent="Import failed: "+e.message}
}
